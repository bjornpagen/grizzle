     // Runs a command, samples its process tree's CPU time with ps every intervalMs, and prints the longest stretch with no CPU progress.
     import { spawn, execFileSync } from "node:child_process"
     const [intervalMs, ...cmd] = process.argv.slice(2)
     const interval = Number(intervalMs)
     const centis = (t) => {
     	const m = /^(?:(\d+)-)?(?:(\d+):)?(\d+):(\d+)(?:\.(\d+))?$/.exec(t.trim())
     	if (!m) throw new Error("bad time " + t)
     	const [, d, h, mi, s, f] = m
     	return ((Number(d ?? 0) * 24 + Number(h ?? 0)) * 60 + Number(mi)) * 6000 + Number(s) * 100 + Number((f ?? "0").padEnd(2, "0").slice(0, 2))
     }
     const treeCpu = (root) => {
     	const out = execFileSync("ps", ["-A", "-o", "pid=,ppid=,time="], { encoding: "utf8" })
     	const rows = out.trim().split("\n").map((l) => l.trim().split(/\s+/))
     	const kids = new Map()
     	const cpu = new Map()
     	for (const [pid, ppid, time] of rows) { cpu.set(Number(pid), centis(time)); const k = kids.get(Number(ppid)) ?? []; k.push(Number(pid)); kids.set(Number(ppid), k) }
     	let total = 0; const stack = [root]; let found = false
     	while (stack.length) { const p = stack.pop(); if (cpu.has(p)) { found = true; total += cpu.get(p) } for (const c of kids.get(p) ?? []) stack.push(c) }
     	return found ? total : undefined
     }
     const t0 = performance.now()
     const child = spawn(cmd[0], cmd.slice(1), { stdio: ["ignore", "ignore", "inherit"] })
     let last = 0, lastProgress = performance.now(), longest = 0, samples = 0, deltas = [], gaps = [], exited = false
     const timer = setInterval(() => {
     	const now = performance.now()
     	if (exited) return
     	const c = treeCpu(child.pid)
     	if (c === undefined || exited) return
     	samples++
     	deltas.push(c - last)
     	if (c > last) { if (now - lastProgress > interval * 1.5) gaps.push([Math.round(lastProgress - t0), Math.round(now - lastProgress)]); last = c; lastProgress = now } else { longest = Math.max(longest, now - lastProgress) }
     }, interval)
     child.on("exit", (code, signal) => {
     	exited = true
     	clearInterval(timer)
     	const wall = performance.now() - t0
     	console.log(JSON.stringify({ cmd: cmd.slice(-3).join(" "), code, signal, wallMs: Math.round(wall), cpuCs: last, samples, longestNoProgressMs: Math.round(longest), gaps, maxDeltaCs: Math.max(...deltas.slice(1)), load: execFileSync("sysctl", ["-n", "vm.loadavg"], { encoding: "utf8" }).trim() }))
     })
